import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader
import pickle
import os
import numpy as np
import matplotlib.pyplot as plt
import cv2
from scipy.special import entr

class MNISTFeatureAnalyzer:
    def __init__(self, model, device=None):
        self.model = model
        self.device = (
            device
            if device
            else torch.device("cuda" if torch.cuda.is_available() else "cpu")
        )
        self.model.to(self.device)

        # Data preprocessing and loading
        transform = transforms.Compose(
            [transforms.ToTensor(), transforms.Normalize((0.5,), (0.5,))]
        )

        self.test_dataset = torchvision.datasets.MNIST(
            root="./data", train=False, transform=transform, download=True
        )

        self.test_loader = DataLoader(
            dataset=self.test_dataset, batch_size=1, shuffle=False
        )

        # Initialize feature maps dictionary
        self.feature_maps = {}

        # Register hooks for convolutional layers
        self._register_conv_hooks()

    def _hook_fn(self, module, input, output):
        """Hook function that only activates in eval mode for convolutional layers"""
        if not self.model.training:  # Ensures it only runs in eval mode
            # Store a copy of the output to avoid any potential issues
            self.feature_maps[module] = (
                output.detach().clone()
            )  # stores activations inside feature_maps dictionary

    def _register_conv_hooks(self):
        """Register forward hooks on all convolutional layers"""
        # Clear any existing hooks first
        self.feature_maps = {}

        for name, module in self.model.named_modules():
            if isinstance(module, (nn.Conv1d, nn.Conv2d, nn.Conv3d)):
                # Remove any existing hooks
                module._forward_hooks.clear()
                # Register new hook
                module.register_forward_hook(self._hook_fn)

    def get_predictions_and_accuracy(self):
        """Get predictions for all test images and calculate per-class accuracy"""
        self.model.eval()
        all_predictions = []
        all_labels = []
        all_indices = []

        with torch.no_grad():
            for idx, (images, labels) in enumerate(self.test_loader):
                images, labels = images.to(self.device), labels.to(self.device)
                outputs = self.model(images)
                _, predicted = torch.max(outputs, 1)

                all_predictions.append(predicted.cpu().item())
                all_labels.append(labels.cpu().item())
                all_indices.append(idx)

        # Calculate per-class accuracy
        class_correct = defaultdict(int)
        class_total = defaultdict(int)
        class_predictions = defaultdict(
            list
        )  # Store (idx, true_label, pred_label) for each class

        for idx, (true_label, pred_label) in enumerate(
            zip(all_labels, all_predictions)
        ):
            class_total[true_label] += 1
            class_predictions[true_label].append((idx, true_label, pred_label))

            if true_label == pred_label:
                class_correct[true_label] += 1

        # Calculate accuracy for each class
        class_accuracy = {}
        for class_label in range(10):
            if class_total[class_label] > 0:
                class_accuracy[class_label] = (
                    class_correct[class_label] / class_total[class_label]
                )
            else:
                class_accuracy[class_label] = 0.0

        return (
            all_predictions,
            all_labels,
            all_indices,
            class_accuracy,
            class_predictions,
        )

    def get_random_predictions_per_class(self, class_predictions, num_samples=3):
        """Get random samples for each class (regardless of prediction correctness)"""
        random_samples = {}

        for class_label in range(10):
            if class_label in class_predictions:
                # Get all predictions for this class
                samples = class_predictions[class_label]
                # Randomly sample up to num_samples
                if len(samples) >= num_samples:
                    random_samples[class_label] = random.sample(samples, num_samples)
                else:
                    random_samples[class_label] = samples

        return random_samples

    def get_misclassified_samples(self, class_predictions, num_samples=3):
        """Get misclassified samples for each class"""
        misclassified_samples = {}

        for class_label in range(10):
            if class_label in class_predictions:
                # Get only misclassified samples for this class
                misclassified = [
                    (idx, true_label, pred_label)
                    for idx, true_label, pred_label in class_predictions[class_label]
                    if true_label != pred_label
                ]

                if len(misclassified) >= num_samples:
                    misclassified_samples[class_label] = random.sample(
                        misclassified, num_samples
                    )
                else:
                    misclassified_samples[class_label] = misclassified

        return misclassified_samples

    def save_feature_maps(self, image_indices, save_dir="./feature_maps"):
        """Save feature maps for given image indices"""
        os.makedirs(save_dir, exist_ok=True)

        saved_paths = []

        for idx in image_indices:
            # Get the image
            image, label = self.test_dataset[idx]
            image = image.unsqueeze(0).to(self.device)  # Add batch dimension

            # Clear previous feature maps
            self.feature_maps.clear()

            # Forward pass to capture feature maps
            with torch.no_grad():
                output = self.model(image)
                _, predicted = torch.max(output, 1)

            # Save feature maps
            save_path = os.path.join(
                save_dir, f"image_{idx}_true_{label}_pred_{predicted.item()}.pkl"
            )

            with open(save_path, "wb") as f:
                pickle.dump(
                    {
                        "feature_maps": {
                            str(layer): fmap.cpu().numpy()
                            for layer, fmap in self.feature_maps.items()
                        },
                        "true_label": label,
                        "predicted_label": predicted.item(),
                        "image_index": idx,
                    },
                    f,
                )

            saved_paths.append(save_path)
            print(
                f"Saved feature maps for image {idx} (True: {label}, Pred: {predicted.item()})"
            )

        return saved_paths

    def compute_PEEK(self, feature_maps, h, w):
        """Compute PEEK map from feature maps"""
        positivized_maps = feature_maps + np.abs(np.min(feature_maps))  # Make positive
        entropy_map = -np.sum(entr(positivized_maps), axis=-1)  # Compute entropy
        peek_map = cv2.resize(
            entropy_map, (w, h)
        )  # Resize to match original image size
        return peek_map

    def save_PEEK_images(self, feature_map_path, save_dir="./peek_images"):
        """Save individual PEEK images for a given feature map file"""
        # Load feature maps
        with open(feature_map_path, "rb") as f:
            data = pickle.load(f)

        feature_maps = data["feature_maps"]
        true_label = data["true_label"]
        predicted_label = data["predicted_label"]
        image_index = data["image_index"]

        # Get original image
        image, _ = self.test_dataset[image_index]
        image_np = image.squeeze().numpy()
        h, w = image_np.shape

        # Get convolutional layers
        conv_layers = [layer for layer in feature_maps.keys() if "Conv" in layer]

        if not conv_layers:
            print(f"No convolutional layers found for image {image_index}")
            return []

        # Create directory structure
        class_dir = os.path.join(save_dir, f"class_{true_label}")
        os.makedirs(class_dir, exist_ok=True)

        saved_paths = []

        # Save original image
        original_path = os.path.join(class_dir, f"image_{image_index}_original.png")
        plt.figure(figsize=(4, 4))
        plt.imshow(image_np, cmap="gray")
        plt.title(f"Original (True: {true_label}, Pred: {predicted_label})")
        plt.axis("off")
        plt.savefig(original_path, dpi=150, bbox_inches="tight")
        plt.close()
        saved_paths.append(original_path)

        # Save PEEK maps for each layer
        for layer_name in conv_layers:
            # Compute PEEK map
            feature_map = feature_maps[layer_name][0]  # Remove batch dimension
            feature_map = np.moveaxis(feature_map, 0, -1)  # Rearrange channels
            peek_map = self.compute_PEEK(feature_map, h, w)

            # Save PEEK map
            peek_path = os.path.join(
                class_dir, f"image_{image_index}_{layer_name}_peek.png"
            )
            plt.figure(figsize=(4, 4))
            plt.imshow(image_np, cmap="gray")  # Original image
            plt.imshow(peek_map, alpha=0.7, cmap="jet")  # Overlay PEEK
            plt.title(f"PEEK - {layer_name}")
            plt.axis("off")
            plt.savefig(peek_path, dpi=150, bbox_inches="tight")
            plt.close()
            saved_paths.append(peek_path)

        print(f"Saved {len(saved_paths)} images for image {image_index} in {class_dir}")
        return saved_paths


def main():
    # Set random seed for reproducibility
    random.seed(42)
    torch.manual_seed(42)

    # Define a simple CNN model (you can replace this with your own model)
    class SimpleCNN(nn.Module):
        def __init__(self):
            super(SimpleCNN, self).__init__()
            # First Conv2D layer: input=1, output=16, kernel=3, stride=1, padding=0
            self.conv1 = nn.Conv2d(1, 16, kernel_size=3, stride=1, padding=0)
            self.relu = nn.ReLU()
            # First MaxPool2D: kernel=2, stride=2, padding=0
            self.pool1 = nn.MaxPool2d(kernel_size=2, stride=2, padding=0)
            # Second Conv2D layer: input=16, output=32, kernel=3, stride=1, padding=0
            self.conv2 = nn.Conv2d(16, 32, kernel_size=3, stride=1, padding=0)
            # Second MaxPool2D: kernel=2, stride=2, padding=0
            self.pool2 = nn.MaxPool2d(kernel_size=2, stride=2, padding=0)
            # Flatten layer
            self.flatten = nn.Flatten(1, -1)
            # First Linear layer: input=800, output=128
            self.fc1 = nn.Linear(800, 128)
            # Second Linear layer: input=128, output=10
            self.fc2 = nn.Linear(128, 10)

        def forward(self, x):
            x = self.conv1(x)
            x = self.relu(x)
            x = self.pool1(x)
            x = self.conv2(x)
            x = self.relu(x)
            x = self.pool2(x)
            x = self.flatten(x)
            x = self.fc1(x)
            x = self.relu(x)
            x = self.fc2(x)
            return x

    # Initialize model and analyzer
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = SimpleCNN()
    model.to(device)  # Move model to device before any operations

    # Load pre-trained model if available
    model_path = "./models/mnist_cnn_model.pth"
    if os.path.exists(model_path):
        model.load_state_dict(torch.load(model_path, map_location=device))
        print(f"Loaded pre-trained model from {model_path}")
    else:
        print("No pre-trained model found. Training a new model...")
        # Train the model (simplified training)
        criterion = nn.CrossEntropyLoss()
        optimizer = optim.Adam(model.parameters(), lr=0.001)

        # Quick training (you might want to train more thoroughly)
        train_dataset = torchvision.datasets.MNIST(
            root="./data",
            train=True,
            transform=transforms.Compose(
                [transforms.ToTensor(), transforms.Normalize((0.5,), (0.5,))]
            ),
            download=True,
        )
        train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)

        model.train()
        for epoch in range(2):
            for batch_idx, (data, target) in enumerate(train_loader):
                data, target = data.to(device), target.to(device)
                optimizer.zero_grad()
                output = model(data)
                loss = criterion(output, target)
                loss.backward()
                optimizer.step()

            print(f"Epoch: {epoch}, Loss: {loss.item():.4f}")

        # Save the trained model
        os.makedirs("./models", exist_ok=True)
        torch.save(model.state_dict(), model_path)
        print(f"Saved trained model to {model_path}")

    # Initialize analyzer
    analyzer = MNISTFeatureAnalyzer(model, device)

    # Get predictions and accuracy
    print("Getting predictions and calculating accuracy...")
    all_predictions, all_labels, all_indices, class_accuracy, class_predictions = (
        analyzer.get_predictions_and_accuracy()
    )

    # Print class accuracy
    print("\nPer-class accuracy:")
    for class_label in range(10):
        print(f"Class {class_label}: {class_accuracy[class_label]:.4f}")

    # Find 3 lowest accuracy classes
    sorted_classes = sorted(class_accuracy.items(), key=lambda x: x[1])
    lowest_accuracy_classes = [
        class_label for class_label, accuracy in sorted_classes[:3]
    ]
    print(f"\n3 lowest accuracy classes: {lowest_accuracy_classes}")

    # Create directory for lowest accuracy classes
    lowest_accuracy_dir = "./lowest_accuracy_classes"
    os.makedirs(lowest_accuracy_dir, exist_ok=True)

    # Get random predictions per class (3 per class)
    print("\nGetting random predictions per class...")
    random_samples = analyzer.get_random_predictions_per_class(
        class_predictions, num_samples=3
    )

    # Get misclassified samples for lowest accuracy classes
    print("Getting misclassified samples for lowest accuracy classes...")
    misclassified_samples = analyzer.get_misclassified_samples(
        class_predictions, num_samples=3
    )

    # Collect all image indices to process
    all_images_to_process = []

    # Add random samples from each class
    for class_label, samples in random_samples.items():
        print(f"\nClass {class_label} - Random samples:")
        for idx, true_label, pred_label in samples:
            print(f"  Image {idx}: True={true_label}, Pred={pred_label}")
            all_images_to_process.append(idx)

    # Add misclassified samples from lowest accuracy classes
    for class_label in lowest_accuracy_classes:
        if class_label in misclassified_samples and misclassified_samples[class_label]:
            print(f"\nClass {class_label} - Misclassified samples:")
            for idx, true_label, pred_label in misclassified_samples[class_label]:
                print(f"  Image {idx}: True={true_label}, Pred={pred_label}")
                all_images_to_process.append(idx)
        else:
            print(f"\nClass {class_label} - No misclassified samples found")

    # Remove duplicates
    all_images_to_process = list(set(all_images_to_process))

    # Save feature maps
    print("\nSaving feature maps...")
    feature_map_paths = analyzer.save_feature_maps(all_images_to_process)

    # Generate PEEK images
    print("\nGenerating PEEK images...")
    peek_image_paths = []
    for feature_map_path in feature_map_paths:
        peek_image_paths.extend(analyzer.save_PEEK_images(feature_map_path))

    # Create summary file for lowest accuracy classes
    summary_path = os.path.join(lowest_accuracy_dir, "lowest_accuracy_summary.txt")
    with open(summary_path, "w") as f:
        f.write("Lowest Accuracy Classes Summary\n")
        f.write("=" * 40 + "\n\n")
        for class_label, accuracy in sorted_classes[:3]:
            f.write(f"Class {class_label}: {accuracy:.4f} accuracy\n")
            if (
                class_label in misclassified_samples
                and misclassified_samples[class_label]
            ):
                f.write("   Misclassified samples:\n")
                for idx, true_label, pred_label in misclassified_samples[class_label]:
                    f.write(f"    Image {idx}: True={true_label}, Pred={pred_label}\n")
            f.write("\n")

    print(f"\nAnalysis complete!")
    print(f"Feature maps saved: {len(feature_map_paths)}")
    print(f"PEEK images generated: {len(peek_image_paths)}")
    print(f"Lowest accuracy summary saved to: {summary_path}")


if __name__ == "__main__":
    main()
